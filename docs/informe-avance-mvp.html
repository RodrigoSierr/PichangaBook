const WEIGHTS = [5, 4, 3, 2, 7, 6, 5, 4, 3, 2];
export function isValidRuc(ruc: string): boolean {
  if (!/^(10|15|17|20)\d{9}$/.test(ruc)) return false;
  const sum = WEIGHTS.reduce((acc, w, i) => acc + w * Number(ruc[i]), 0);
  let check = 11 - (sum % 11);
  if (check === 10) check = 0; else if (check === 11) check = 1;
  return check === Number(ruc[10]);
}

@WebSocketGateway({ cors: { origin: '*', credentials: true }, transports: ['websocket', 'polling'] })
export class BookingsGateway implements OnGatewayConnection, OnModuleInit {
  @WebSocketServer() server: Server;

  onModuleInit(): void {
    this.sub = this.bus.stream$.subscribe(({ rooms, event, payload }) => {
      for (const room of rooms) this.server.to(room).emit(event, payload);
      this.server.to('demo:global').emit(event, payload);
    });
  }

  @SubscribeMessage('slot:block')
  async onSlotBlock(@MessageBody() dto: ReceptionBlockDto) {
    const slot = await this.bookings.receptionBlock(dto);
    return { ok: true, slot };
  }
}

signClipUpload(userId: string) {
  const timestamp = Math.floor(Date.now() / 1000);
  const params = {
    folder: 'pichangabook/clips',
    public_id: `sub_${userId}_${timestamp}`,
    eager: `so_0,eo_${CLIP_MAX_SECONDS},q_auto:eco,w_480,vc_auto/mp4`,
    timestamp,
  };
  const toSign = Object.keys(params).sort().map((k) => `${k}=${params[k]}`).join('&');
  const signature = createHash('sha1').update(toSign + (apiSecret || 'demo-secret')).digest('hex');
  return { uploadUrl: `https://api.cloudinary.com/v1_1/${cloudName}/video/upload`, signature, ...params };
}

1. Autenticación Segura OTP por SMS y Registro por Rol

  • Mitigación de Trolls y Reservas Falsas: Ningún usuario puede reservar ni publicar canchas sin validar su línea móvil mediante un código de un solo uso (OTP de 6 dígitos con expiración de 5 min y límite de 5 intentos en OtpService).

2. Protección contra Extorsión: Enmascaramiento E.164 y Redacción Automática
POST /api/substitutes/:id/invite POST /api/relays/:id/messages Columna Generada: phone_masked
  • Enmascaramiento en Base de Datos: La API jamás serializa phone_e164; solo expone la columna calculada phone_masked (+51 9•• ••• 101).
  • Central Proxy VoIP y Sanitización de Mensajes: Al convocar a un suplente se asigna un número virtual temporal (+51 (054) 64XX-XXX) con vigencia de 4 horas y el filtro redactContactInfo() bloquea cualquier intento de filtrar números celulares o correos en el chat.

-- En schema.sql (Columna generada almacenada en PostgreSQL):
phone_masked VARCHAR(24) GENERATED ALWAYS AS (
  left(phone_e164, 3) || ' ' || substr(phone_e164, 4, 1) || '•• ••• ' || right(phone_e164, 3)
) STORED

// En src/common/utils/phone.util.ts (Filtro antifraude en mensajes del chat interno):
const PHONE_IN_TEXT = /(\+?\s?51[\s.-]?)?\b9(?:[\s.-]?\d){8}\b|\b\d(?:[\s.-]?\d){8,13}\b/g;
export function redactContactInfo(text: string): { body: string; redacted: boolean } {
  let redacted = false;
  const body = text.replace(PHONE_IN_TEXT, () => { redacted = true; return '[número oculto 🔒]'; });
  return { body, redacted };
}
